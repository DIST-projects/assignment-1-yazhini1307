Q1. Implement a deque using a doubly linked list
#include <stdio.h>
#include <stdlib.h>

struct Node {
    int data;
    struct Node *prev, *next;
};

struct Node *front = NULL, *rear = NULL;

void insertFront(int x)
{
    struct Node *n = malloc(sizeof(struct Node));
    n->data = x;
    n->prev = NULL;
    n->next = front;

    if (front) front->prev = n;
    else rear = n;

    front = n;
}

void insertRear(int x)
{
    struct Node *n = malloc(sizeof(struct Node));
    n->data = x;
    n->next = NULL;
    n->prev = rear;

    if (rear) rear->next = n;
    else front = n;

    rear = n;
}

void deleteFront()
{
    struct Node *p;

    if (!front) return;

    p = front;
    front = front->next;

    if (front) front->prev = NULL;
    else rear = NULL;

    free(p);
}

void deleteRear()
{
    struct Node *p;

    if (!rear) return;

    p = rear;
    rear = rear->prev;

    if (rear) rear->next = NULL;
    else front = NULL;

    free(p);
}

int main()
{
    int q, op, x, i;
    struct Node *p;

    scanf("%d", &q);

    for (i = 0; i < q; i++) {
        scanf("%d", &op);

        if (op == 1) {
            scanf("%d", &x);
            insertFront(x);
        }
        else if (op == 2) {
            scanf("%d", &x);
            insertRear(x);
        }
        else if (op == 3) deleteFront();
        else if (op == 4) deleteRear();
        else if (op == 5) {
            for (p = front; p; p = p->next)
                printf("%d ", p->data);
            printf("\n");
        }
    }

    return 0;
}

Q2. Find the intersection of two sorted lists
#include <stdio.h>

int main()
{
    int a[100], b[100], n, m, i = 0, j = 0;

    scanf("%d", &n);
    for (i = 0; i < n; i++)
        scanf("%d", &a[i]);

    scanf("%d", &m);
    for (i = 0; i < m; i++)
        scanf("%d", &b[i]);

    i = j = 0;

    while (i < n && j < m) {
        if (a[i] == b[j]) {
            printf("%d ", a[i]);
            i++;
            j++;
        }
        else if (a[i] < b[j])
            i++;
        else
            j++;
    }

    printf("\n");
    return 0;
}

Q3. Swap two positions in a list
#include <stdio.h>

int main()
{
    int a[100], n, x, y, i, t;

    scanf("%d", &n);
    if (n < 1 || n > 100) return 0;

    for (i = 0; i < n; i++)
        scanf("%d", &a[i]);

    scanf("%d%d", &x, &y);

    if (x >= 1 && x <= n && y >= 1 && y <= n) {
        t = a[x - 1];
        a[x - 1] = a[y - 1];
        a[y - 1] = t;
    }

    for (i = 0; i < n; i++)
        printf("%d ", a[i]);

    printf("\n");
    return 0;
}

Q4. Self-adjusting list: move the accessed element to the front
#include <stdio.h>

int main()
{
    int a[100], n, x, i, j, t;

    scanf("%d", &n);
    if (n < 0 || n > 100) return 0;

    for (i = 0; i < n; i++)
        scanf("%d", &a[i]);

    scanf("%d", &x);

    for (i = 0; i < n; i++)
        if (a[i] == x) break;

    if (i < n) {
        t = a[i];

        for (j = i; j > 0; j--)
            a[j] = a[j - 1];

        a[0] = t;
    }

    for (i = 0; i < n; i++)
        printf("%d ", a[i]);

    printf("\n");
    return 0;
}

Q5. Reverse a list
#include <stdio.h>

int main()
{
    int a[100], n, i, t;

    scanf("%d", &n);
    if (n < 0 || n > 100) return 0;

    for (i = 0; i < n; i++)
        scanf("%d", &a[i]);

    for (i = 0; i < n / 2; i++) {
        t = a[i];
        a[i] = a[n - i - 1];
        a[n - i - 1] = t;
    }

    for (i = 0; i < n; i++)
        printf("%d ", a[i]);

    printf("\n");
    return 0;
}

Q6. Delete the middle element of a list
#include <stdio.h>

int main()
{
    int a[100], n, i, mid;

    scanf("%d", &n);
    if (n < 0 || n > 100) return 0;

    for (i = 0; i < n; i++)
        scanf("%d", &a[i]);

    if (n > 0) {
        mid = n / 2;

        for (i = mid; i < n - 1; i++)
            a[i] = a[i + 1];

        n--;
    }

    for (i = 0; i < n; i++)
        printf("%d ", a[i]);

    printf("\n");
    return 0;
}
