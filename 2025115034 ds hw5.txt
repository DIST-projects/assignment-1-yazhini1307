Q1. Polynomial addition and subtraction using linked list
#include <stdio.h>
#include <stdlib.h>

struct Node {
    int c, p;
    struct Node *next;
};

void add(struct Node **h, int c, int p)
{
    struct Node *t = *h, *n;

    while (t && t->p != p)
        t = t->next;

    if (t) {
        t->c += c;
        return;
    }

    n = malloc(sizeof(struct Node));
    n->c = c;
    n->p = p;
    n->next = *h;
    *h = n;
}

void display(struct Node *t)
{
    if (!t) {
        printf("0\n");
        return;
    }

    while (t) {
        printf("%dx^%d", t->c, t->p);
        if (t->next) printf(" + ");
        t = t->next;
    }
    printf("\n");
}

int main()
{
    struct Node *a = NULL, *b = NULL;
    struct Node *s = NULL, *d = NULL, *t;
    int n, m, c, p, i;

    scanf("%d", &n);
    for (i = 0; i < n; i++) {
        scanf("%d%d", &c, &p);
        add(&a, c, p);
    }

    scanf("%d", &m);
    for (i = 0; i < m; i++) {
        scanf("%d%d", &c, &p);
        add(&b, c, p);
    }

    for (t = a; t; t = t->next) {
        add(&s, t->c, t->p);
        add(&d, t->c, t->p);
    }

    for (t = b; t; t = t->next) {
        add(&s, t->c, t->p);
        add(&d, -t->c, t->p);
    }

    display(s);
    display(d);

    return 0;
}

Q2. Find the previous node in a circular linked list
#include <stdio.h>
#include <stdlib.h>

struct Node {
    int data;
    struct Node *next;
};

int main()
{
    struct Node *head = NULL, *tail = NULL;
    struct Node *n, *p;
    int count, x, i;

    scanf("%d", &count);

    for (i = 0; i < count; i++) {
        scanf("%d", &x);
        n = malloc(sizeof(struct Node));
        n->data = x;

        if (!head)
            head = tail = n;
        else {
            tail->next = n;
            tail = n;
        }

        tail->next = head;
    }

    scanf("%d", &x);

    if (!head) {
        printf("-1\n");
        return 0;
    }

    p = head;

    do {
        if (p->next->data == x) {
            printf("%d\n", p->data);
            return 0;
        }
        p = p->next;
    } while (p != head);

    printf("-1\n");
    return 0;
}

Q3. Insert an element at a given position in a circular linked list
#include <stdio.h>
#include <stdlib.h>

struct Node {
    int data;
    struct Node *next;
};

int main()
{
    struct Node *head = NULL, *tail = NULL;
    struct Node *n, *p;
    int count, x, pos, i;

    scanf("%d", &count);

    for (i = 0; i < count; i++) {
        scanf("%d", &x);
        n = malloc(sizeof(struct Node));
        n->data = x;

        if (!head)
            head = tail = n;
        else {
            tail->next = n;
            tail = n;
        }

        tail->next = head;
    }

    scanf("%d%d", &x, &pos);

    if (pos < 1) return 0;

    n = malloc(sizeof(struct Node));
    n->data = x;

    if (!head) {
        head = tail = n;
        n->next = n;
    }
    else if (pos == 1) {
        n->next = head;
        head = n;
        tail->next = head;
    }
    else {
        p = head;

        for (i = 1; i < pos - 1 && p->next != head; i++)
            p = p->next;

        n->next = p->next;
        p->next = n;

        if (p == tail)
            tail = n;
    }

    p = head;

    do {
        printf("%d ", p->data);
        p = p->next;
    } while (p != head);

    printf("\n");
    return 0;
}

Q6. Implement two stacks using one array
#include <stdio.h>

#define SIZE 100

int a[SIZE], top1 = -1, top2 = SIZE;

void push1(int x)
{
    if (top1 + 1 < top2)
        a[++top1] = x;
}

void push2(int x)
{
    if (top1 + 1 < top2)
        a[--top2] = x;
}

int main()
{
    int n, m, i, x;

    scanf("%d", &n);
    if (n < 0 || n > SIZE) return 0;

    for (i = 0; i < n; i++) {
        scanf("%d", &x);
        push1(x);
    }

    scanf("%d", &m);
    if (m < 0 || m > SIZE) return 0;

    for (i = 0; i < m; i++) {
        scanf("%d", &x);
        push2(x);
    }

    for (i = top1; i >= 0; i--)
        printf("%d ", a[i]);
    printf("\n");

    for (i = top2; i < SIZE; i++)
        printf("%d ", a[i]);
    printf("\n");

    return 0;
}

Q7. Flatten a linked list
Each node has next and bottom pointers. Each bottom-linked list is sorted. This program merges them into one sorted list using the bottom pointer.
#include <stdio.h>
#include <stdlib.h>

struct Node {
    int data;
    struct Node *next, *bottom;
};

struct Node *merge(struct Node *a, struct Node *b)
{
    struct Node *r = NULL;

    if (!a) return b;
    if (!b) return a;

    if (a->data <= b->data) {
        r = a;
        r->bottom = merge(a->bottom, b);
    }
    else {
        r = b;
        r->bottom = merge(a, b->bottom);
    }

    r->next = NULL;
    return r;
}

struct Node *flatten(struct Node *root)
{
    if (!root || !root->next)
        return root;

    root->next = flatten(root->next);
    root = merge(root, root->next);

    return root;
}

int main()
{
    struct Node *a, *b, *c, *p;

    a = malloc(sizeof(struct Node));
    b = malloc(sizeof(struct Node));
    c = malloc(sizeof(struct Node));

    a->data = 5; a->next = b;
    b->data = 10; b->next = c;
    c->data = 19; c->next = NULL;

    a->bottom = malloc(sizeof(struct Node));
    a->bottom->data = 7;
    a->bottom->bottom = malloc(sizeof(struct Node));
    a->bottom->bottom->data = 8;
    a->bottom->bottom->bottom = NULL;

    b->bottom = malloc(sizeof(struct Node));
    b->bottom->data = 20;
    b->bottom->bottom = NULL;

    c->bottom = malloc(sizeof(struct Node));
    c->bottom->data = 22;
    c->bottom->bottom = NULL;

    a = flatten(a);

    for (p = a; p; p = p->bottom)
        printf("%d ", p->data);

    printf("\n");
    return 0;
}

Q8. Merge two sorted linked lists
#include <stdio.h>
#include <stdlib.h>

struct Node {
    int data;
    struct Node *next;
};

void insert(struct Node **h, int x)
{
    struct Node *n = malloc(sizeof(struct Node)), *p;
    n->data = x;
    n->next = NULL;

    if (!*h) {
        *h = n;
        return;
    }

    p = *h;
    while (p->next) p = p->next;
    p->next = n;
}

int main()
{
    struct Node *a = NULL, *b = NULL, *c = NULL;
    struct Node *p;
    int n, m, x, i;

    scanf("%d", &n);
    for (i = 0; i < n; i++) {
        scanf("%d", &x);
        insert(&a, x);
    }

    scanf("%d", &m);
    for (i = 0; i < m; i++) {
        scanf("%d", &x);
        insert(&b, x);
    }

    while (a && b) {
        if (a->data <= b->data) {
            insert(&c, a->data);
            a = a->next;
        }
        else {
            insert(&c, b->data);
            b = b->next;
        }
    }

    while (a) {
        insert(&c, a->data);
        a = a->next;
    }

    while (b) {
        insert(&c, b->data);
        b = b->next;
    }

    for (p = c; p; p = p->next)
        printf("%d ", p->data);

    printf("\n");
    return 0;
}

Q9. Swap the kth element from the beginning with the kth element from the end

int main()
{
    int a[1000], n, k, i, j, t;

    scanf("%d", &n);
    if (n < 1 || n > 1000) return 0;

    for (i = 0; i < n; i++)
        scanf("%d", &a[i]);

    scanf("%d", &k);

    if (k < 1 || k > n) return 0;

    i = k - 1;
    j = n - k;

    t = a[i];
    a[i] = a[j];

    printf("\n");
    return 0;
}
