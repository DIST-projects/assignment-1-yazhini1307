Q1. BFS and DFS traversal of a graph
#include <stdio.h>

int g[50][50], vis[50], n;

void dfs(int u)
{
    int v;
    vis[u] = 1;
    printf("%d ", u);

    for (v = 0; v < n; v++)
        if (g[u][v] && !vis[v])
            dfs(v);
}

void bfs(int s)
{
    int q[50], seen[50] = {0};
    int f = 0, r = 0, u, v;

    q[r++] = s;
    seen[s] = 1;

    while (f < r) {
        u = q[f++];
        printf("%d ", u);

        for (v = 0; v < n; v++)
            if (g[u][v] && !seen[v]) {
                seen[v] = 1;
                q[r++] = v;
            }
    }
}

int main()
{
    int e, i, u, v;

    scanf("%d%d", &n, &e);
    if (n < 1 || n > 50 || e < 0) return 0;

    for (i = 0; i < e; i++) {
        scanf("%d%d", &u, &v);

        if (u >= 0 && u < n && v >= 0 && v < n)
            g[u][v] = g[v][u] = 1;
    }

    bfs(0);
    printf("\n");

    for (i = 0; i < n; i++) vis[i] = 0;

    dfs(0);
    printf("\n");

    return 0;
}

Q2. Prim's minimum spanning tree
#include <stdio.h>

#define INF 999999

int main()
{
    int g[50][50], used[50] = {0};
    int n, i, j, k, v, min, cost = 0;

    scanf("%d", &n);
    if (n < 1 || n > 50) return 0;

    for (i = 0; i < n; i++)
        for (j = 0; j < n; j++)
            scanf("%d", &g[i][j]);

    used[0] = 1;

    for (i = 1; i < n; i++) {
        min = INF;
        v = -1;

        for (j = 0; j < n; j++)
            if (used[j])
                for (k = 0; k < n; k++)
                    if (!used[k] && g[j][k] > 0 &&
                        g[j][k] < min) {
                        min = g[j][k];
                        v = k;
                    }

        if (v == -1) {
            printf("-1\n");
            return 0;
        }

        used[v] = 1;
        cost += min;
    }

    printf("%d\n", cost);
    return 0;
}

Q3. Kruskal's minimum spanning tree
#include <stdio.h>
#include <stdlib.h>

struct Edge {
    int u, v, w;
} e[1000];

int p[100];

int find(int x)
{
    if (p[x] == x) return x;
    return p[x] = find(p[x]);
}

int cmp(const void *a, const void *b)
{
    return ((struct Edge *)a)->w -
           ((struct Edge *)b)->w;
}

int main()
{
    int n, m, i, a, b, count = 0, cost = 0;

    scanf("%d%d", &n, &m);
    if (n < 1 || n > 100 || m < 0 || m > 1000)
        return 0;

    for (i = 0; i < m; i++)
        scanf("%d%d%d", &e[i].u, &e[i].v, &e[i].w);

    for (i = 0; i < n; i++) p[i] = i;

    qsort(e, m, sizeof(e[0]), cmp);

    for (i = 0; i < m && count < n-1; i++) {
        a = find(e[i].u);
        b = find(e[i].v);

        if (a != b) {
            p[a] = b;
            cost += e[i].w;
            count++;
        }
    }

    printf("%d\n", count == n-1 ? cost : -1);
    return 0;
}

Q4. Dijkstra's shortest path algorithm
#include <stdio.h>

#define INF 999999

int main()
{
    int g[50][50], d[50], used[50] = {0};
    int n, s, i, j, u, min;

    scanf("%d", &n);
    if (n < 1 || n > 50) return 0;

    for (i = 0; i < n; i++)
        for (j = 0; j < n; j++)
            scanf("%d", &g[i][j]);

    scanf("%d", &s);
    if (s < 0 || s >= n) return 0;

    for (i = 0; i < n; i++) d[i] = INF;
    d[s] = 0;

    for (i = 0; i < n; i++) {
        u = -1;
        min = INF;

        for (j = 0; j < n; j++)
            if (!used[j] && d[j] < min) {
                min = d[j];
                u = j;
            }

        if (u == -1) break;

        used[u] = 1;

        for (j = 0; j < n; j++)
            if (g[u][j] > 0 &&
                d[u] + g[u][j] < d[j])
                d[j] = d[u] + g[u][j];
    }

    for (i = 0; i < n; i++) {
        if (d[i] == INF) printf("INF ");
        else printf("%d ", d[i]);
    }

    printf("\n");
    return 0;
}

Q5. Find a mother vertex in a directed graph
#include <stdio.h>

int g[50][50], vis[50], n;

void dfs(int u)
{
    int v;
    vis[u] = 1;

    for (v = 0; v < n; v++)
        if (g[u][v] && !vis[v])
            dfs(v);
}

int main()
{
    int e, i, u, v, last = 0;

    scanf("%d%d", &n, &e);
    if (n < 1 || n > 50 || e < 0) return 0;

    for (i = 0; i < e; i++) {
        scanf("%d%d", &u, &v);

        if (u >= 0 && u < n && v >= 0 && v < n)
            g[u][v] = 1;
    }

    for (i = 0; i < n; i++)
        if (!vis[i]) {
            dfs(i);
            last = i;
        }

    for (i = 0; i < n; i++) vis[i] = 0;

    dfs(last);

    for (i = 0; i < n; i++)
        if (!vis[i]) {
            printf("-1\n");
            return 0;
        }

    printf("%d\n", last);
    return 0;
}

Q6. Check whether a graph is bipartite
#include <stdio.h>

int g[50][50], col[50], n;

int dfs(int u, int c)
{
    int v;
    col[u] = c;

    for (v = 0; v < n; v++) {
        if (g[u][v]) {
            if (col[v] == -1) {
                if (!dfs(v, 1-c)) return 0;
            }
            else if (col[v] == c)
                return 0;
        }
    }

    return 1;
}

int main()
{
    int e, i, u, v;

    scanf("%d%d", &n, &e);
    if (n < 1 || n > 50 || e < 0) return 0;

    for (i = 0; i < n; i++) col[i] = -1;

    for (i = 0; i < e; i++) {
        scanf("%d%d", &u, &v);

        if (u >= 0 && u < n && v >= 0 && v < n)
            g[u][v] = g[v][u] = 1;
    }

    for (i = 0; i < n; i++)
        if (col[i] == -1 && !dfs(i, 0)) {
            printf("Not Bipartite\n");
            return 0;
        }

    printf("Bipartite\n");
    return 0;
}

Q7. Topological sort
#include <stdio.h>

int g[50][50], in[50];

int main()
{
    int n, e, i, u, v, q[50];
    int f = 0, r = 0, count = 0;

    scanf("%d%d", &n, &e);
    if (n < 1 || n > 50 || e < 0) return 0;

    for (i = 0; i < e; i++) {
        scanf("%d%d", &u, &v);

        if (u >= 0 && u < n && v >= 0 && v < n &&
            !g[u][v]) {
            g[u][v] = 1;
            in[v]++;
        }
    }

    for (i = 0; i < n; i++)
        if (in[i] == 0)
            q[r++] = i;

    while (f < r) {
        u = q[f++];
        printf("%d ", u);
        count++;

        for (v = 0; v < n; v++)
            if (g[u][v]) {
                in[v]--;
                if (in[v] == 0)
                    q[r++] = v;
            }
    }

    if (count < n) printf("Cycle Found");

    printf("\n");
    return 0;
}

Q8. Hashing using linear probing
#include <stdio.h>

int main()
{
    int h[7], a[] = {76, 93, 40, 47, 10, 55};
    int i, j, k;

    for (i = 0; i < 7; i++) h[i] = -1;

    for (i = 0; i < 6; i++) {
        k = a[i] % 7;

        for (j = 0; j < 7; j++)
            if (h[(k+j)%7] == -1) {
                h[(k+j)%7] = a[i];
                break;
            }
    }

    for (i = 0; i < 7; i++)
        printf("%d : %d\n", i, h[i]);

    return 0;
}
